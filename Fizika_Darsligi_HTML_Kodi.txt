<script>
    const curriculum = [
        {
            chapter: "I BOB. ELEKTR ZARYAD. ELEKTR MAYDON",
            topics: [
                { id: 1, title: "1-§. Jismlarning zaryadlanishi", type: "charge_sim" },
                { id: 3, title: "3-§. Zaryadlarning o'zaro ta'siri. Kulon qonuni", type: "coulomb_sim" }
            ]
        },
        {
            chapter: "II BOB. ELEKTR TOKI",
            topics: [
                { id: 17, title: "17-§. Zanjirning bir qismi uchun Om qonuni", type: "ohm_sim" }
            ]
        }
    ];

    let currentActiveId = 1;

    function initNavigation() {
        const navList = document.getElementById('navList');
        navList.innerHTML = '';
        
        curriculum.forEach(chap => {
            const chapDiv = document.createElement('div');
            chapDiv.className = 'chapter-title';
            chapDiv.innerText = chap.chapter;
            navList.appendChild(chapDiv);

            chap.topics.forEach(top => {
                const item = document.createElement('div');
                item.className = `nav-item ${top.id === currentActiveId ? 'active' : ''}`;
                item.innerText = top.title;
                item.onclick = () => loadTopic(top.id, chap.chapter);
                navList.appendChild(item);
            });
        });
    }

    function loadTopic(id, chapterName) {
        currentActiveId = id;
        initNavigation();
        
        let foundTopic = null;
        curriculum.forEach(c => {
            let t = c.topics.find(tp => tp.id === id);
            if(t) foundTopic = t;
        });

        if(!foundTopic) return;

        document.getElementById('pageTitle').innerText = foundTopic.title;
        document.getElementById('currentChapterIndicator').innerText = chapterName;
        
        const contentBody = document.getElementById('mainContent');
        contentBody.innerHTML = renderTopicContent(foundTopic);
        
        initSimulation(foundTopic.type);
        if(window.MathJax) {
            MathJax.typeset();
        }
    }

    function renderTopicContent(topic) {
        switch(topic.type) {
            case 'charge_sim':
                return `
                    <div class="card">
                        <h2>Mavzu tahlili va nazariya</h2>
                        <p>Tabiatda ikki xil elektr zaryadi mavjud: musbat ($+$) va manfiy ($-$). Ishqalanish natijasida jismlar elektrlanadi. Bir xil ishorali zaryadlar bir-birini itaradi, har xil ishorali zaryadlar esa tortishadi.</p>
                        <div class="formula-box">
                            Zaryad birligi: Kulon (Kl). Elementar zaryad: $e = 1.6 \\times 10^{-19} \\text{ Kl}$
                        </div>
                    </div>
                    <div class="card">
                        <h2>Interaktiv Simulyatsiya: Zaryadlarning o'zaro ta'siri</h2>
                        <div class="sim-box">
                            <canvas id="simCanvas" width="650" height="250"></canvas>
                            <div class="controls">
                                <label>1-zaryad:</label>
                                <select id="charge1Type" onchange="resetSimState()">
                                    <option value="+">Musbat (+)</option>
                                    <option value="-">Manfiy (-)</option>
                                </select>
                                <label>2-zaryad:</label>
                                <select id="charge2Type" onchange="resetSimState()">
                                    <option value="+" selected>Musbat (+)</option>
                                    <option value="-">Manfiy (-)</option>
                                </select>
                                <button class="primary" onclick="resetSimState()">Qayta boshlash</button>
                            </div>
                        </div>
                    </div>`;
            case 'coulomb_sim':
                return `
                    <div class="card">
                        <h2>Kulon qonuni</h2>
                        <p>Ikkita qo'zg'almas nuqtaviy zaryadlarning o'zaro ta'sir kuchi zaryadlar ko'paytmasiga to'g'ri proporsional, ular orasidagi masofa kvadratiga esa teskari proporsionaldir.</p>
                        <div class="formula-box">
                            $$F = k \\cdot \\frac{|q_1 \\cdot q_2|}{r^2}$$
                        </div>
                    </div>
                    <div class="card">
                        <h2>Simulyatsiya: Masofa va Kuch bog'liqligi</h2>
                        <div class="sim-box">
                            <canvas id="simCanvas" width="650" height="220"></canvas>
                            <div class="controls">
                                <label>Masofa ($r$): <span id="distVal">150</span>px</label>
                                <input type="range" id="distRange" min="50" max="280" value="150" oninput="updateCoulomb()">
                            </div>
                        </div>
                    </div>`;
            case 'ohm_sim':
                return `
                    <div class="card">
                        <h2>Zanjirning bir qismi uchun Om qonuni</h2>
                        <p>O'tkazgichdan o'tayotgan tok kuchi uning uchlaridagi kuchlanishga to'g'ri proporsional, qarshiligiga esa teskari proporsionaldir.</p>
                        <div class="formula-box">
                            $$I = \\frac{U}{R}$$
                        </div>
                    </div>
                    <div class="card">
                        <h2>Oqim, Kuchlanish va Qarshilik interaktiv modeli</h2>
                        <div class="sim-box">
                            <canvas id="simCanvas" width="650" height="220"></canvas>
                            <div class="controls">
                                <label>Kuchlanish ($U$): <span id="uVal">12</span> V</label>
                                <input type="range" id="uRange" min="2" max="50" value="12" oninput="updateOhm()">
                                <label>Qarshilik ($R$): <span id="rVal">10</span> $\\Omega$</label>
                                <input type="range" id="rRange" min="2" max="50" value="10" oninput="updateOhm()">
                            </div>
                        </div>
                    </div>`;
            default:
                return `<div class="card"><h2>Mavzu</h2><p>Tez kunda qo'shiladi.</p></div>`;
        }
    }

    let simAnimId = null;
    function initSimulation(type) {
        if(simAnimId) cancelAnimationFrame(simAnimId);
        const canvas = document.getElementById('simCanvas');
        if(!canvas) return;
        const ctx = canvas.getContext('2d');

        if(type === 'charge_sim') {
            let x1 = 180, x2 = 470;
            let vx1 = 0, vx2 = 0;
            
            function loop() {
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                let c1 = document.getElementById('charge1Type').value;
                let c2 = document.getElementById('charge2Type').value;

                let dx = x2 - x1;
                let dist = Math.abs(dx);
                let sameSign = (c1 === c2);
                
                if(dist > 50 && dist < 450) {
                    let force = 600 / (dist * dist);
                    if(sameSign) {
                        vx1 -= force * 0.25;
                        vx2 += force * 0.25;
                    } else {
                        vx1 += force * 0.25;
                        vx2 -= force * 0.25;
                    }
                }
                vx1 *= 0.90; vx2 *= 0.90;
                x1 += vx1; x2 += vx2;

                if(x1 < 30) x1 = 30;
                if(x2 > canvas.width - 30) x2 = canvas.width - 30;

                ctx.beginPath();
                ctx.arc(x1, 125, 25, 0, Math.PI*2);
                ctx.fillStyle = c1 === '+' ? '#e74c3c' : '#3498db';
                ctx.fill();
                ctx.fillStyle = 'white'; ctx.font = 'bold 22px Arial'; ctx.textAlign='center'; ctx.textBaseline='middle';
                ctx.fillText(c1, x1, 125);

                ctx.beginPath();
                ctx.arc(x2, 125, 25, 0, Math.PI*2);
                ctx.fillStyle = c2 === '+' ? '#e74c3c' : '#3498db';
                ctx.fill();
                ctx.fillStyle = 'white';
                ctx.fillText(c2, x2, 125);

                simAnimId = requestAnimationFrame(loop);
            }
            loop();
        } else if(type === 'coulomb_sim') {
            window.updateCoulomb = function() {
                let r = document.getElementById('distRange').value;
                document.getElementById('distVal').innerText = r;
                ctx.clearRect(0, 0, canvas.width, canvas.height);
                
                let x1 = canvas.width/2 - parseInt(r)/2;
                let x2 = canvas.width/2 + parseInt(r)/2;

                ctx.beginPath(); ctx.arc(x1, 110, 22, 0, Math.PI*2); ctx.fillStyle='#e74c3c'; ctx.fill();
                ctx.fillStyle='white'; ctx.textAlign='center'; ctx.textBaseline='middle'; ctx.fillText('+', x1, 110);
                
                ctx.beginPath(); ctx.arc(x2, 110, 22, 0, Math.PI*2); ctx.fillStyle='#e74c3c'; ctx.fill();
                ctx.fillStyle='white'; ctx.fillText('+', x2, 110);

                ctx.fillStyle = '#2c3e50'; ctx.font='13px Arial'; ctx.textAlign='left';
                ctx.fillText(`Masofa r = ${r} px | Kuch masofaga teskari kvadrat bog'liq`, 20, 25);
            }
            updateCoulomb();
        } else if(type === 'ohm_sim') {
            window.updateOhm = function() {
                let U = parseFloat(document.getElementById('uRange').value);
                let R = parseFloat(document.getElementById('rRange').value);
                document.getElementById('uVal').innerText = U;
                document.getElementById('rVal').innerText = R;

                let I = (U / R).toFixed(2);
                ctx.clearRect(0, 0, canvas.width, canvas.height);

                ctx.fillStyle = '#2c3e50';
                ctx.font = '15px Arial';
                ctx.textAlign = 'left';
                ctx.fillText(`Tok kuchi: I = U / R = ${U}V / ${R}Ω = ${I} A`, 20, 30);

                ctx.beginPath();
                ctx.arc(canvas.width - 120, 110, 35, 0, Math.PI*2);
                ctx.fillStyle = `rgba(241, 196, 15, ${Math.min(I / 4, 1)})`;
                ctx.fill();
                ctx.lineWidth = 2; ctx.strokeStyle = '#f39c12'; ctx.stroke();
                ctx.fillStyle = '#333'; ctx.font='12px Arial'; ctx.textAlign='center';
                ctx.fillText("Iste'molchi", canvas.width - 120, 160);
            }
            updateOhm();
        }
    }

    function resetSimState() {
        initSimulation('charge_sim');
    }

    initNavigation();
    loadTopic(1, "I BOB. ELEKTR ZARYAD. ELEKTR MAYDON");
</script>
